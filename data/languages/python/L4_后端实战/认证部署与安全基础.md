## 概念讲解

### 为什么安全如此重要？

想象你开了一家实体店，安全措施包括：门锁（认证）、监控摄像头（日志）、保险柜（加密存储）。Web 应用也一样，需要层层防护。

**Web 安全三大目标：**
- **机密性**：只有授权的人能看到数据（比如用户密码）
- **完整性**：数据不能被篡改（比如订单金额）
- **可用性**：服务始终能正常访问（不被攻击打垮）

### 常见攻击类型（用生活比喻理解）

| 攻击类型 | 生活比喻 | 攻击方式 |
|---------|---------|---------|
| **XSS** | 有人在你的留言板上贴了"自动转发所有消息给骗子"的纸条 | 注入恶意脚本到网页 |
| **SQL 注入** | 有人在你店里的搜索框输入"删除所有商品"的指令 | 在输入框注入 SQL 代码 |
| **CSRF** | 有人伪造你的签名去银行转账 | 诱导你浏览器发起恶意请求 |

### 认证与密码安全

用户密码就像保险柜钥匙，**绝不能明文存储**！我们使用**哈希**（单向加密）来存储密码。

```
明文密码 → 哈希函数 → 哈希值（不可逆）
"mypassword123" → "5e884898da28047151d0e56f8dc62927..."
```

即使数据库泄露，攻击者也无法从哈希值反推出原始密码。

## 示例 1：安全的密码存储

```python
# 使用 hashlib 和 secrets 实现安全的密码哈希
import hashlib
import secrets

def hash_password(password: str) -> tuple:
    """生成密码哈希和盐值"""
    # 生成随机盐值（16字节）
    salt = secrets.token_hex(16)
    # 使用 PBKDF2 算法（比简单 SHA256 更安全）
    hash_value = hashlib.pbkdf2_hmac(
        'sha256',           # 哈希算法
        password.encode(),  # 密码转字节
        salt.encode(),      # 盐值转字节
        100000              # 迭代次数（越大越安全，但越慢）
    ).hex()
    return salt, hash_value

def verify_password(password: str, salt: str, expected_hash: str) -> bool:
    """验证密码是否正确"""
    hash_value = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode(),
        salt.encode(),
        100000
    ).hex()
    return hash_value == expected_hash

# 使用示例
if __name__ == "__main__":
    # 用户注册时
    salt, hash_val = hash_password("my_secure_password")
    print(f"盐值: {salt[:8]}...")
    print(f"哈希值: {hash_val[:16]}...")
    
    # 用户登录时
    is_correct = verify_password("my_secure_password", salt, hash_val)
    print(f"密码验证结果: {is_correct}")
    
    is_wrong = verify_password("wrong_password", salt, hash_val)
    print(f"错误密码验证结果: {is_wrong}")

# 预期输出：
# 盐值: 3f9a2b8c...
# 哈希值: 5e884898...
# 密码验证结果: True
# 错误密码验证结果: False
```

## 示例 2：防御 XSS 攻击

```python
# 演示 XSS 攻击和防御
from html import escape

class CommentSystem:
    def __init__(self):
        self.comments = []
    
    def add_comment_unsafe(self, content: str) -> str:
        """不安全版本：直接存储用户输入"""
        self.comments.append(content)
        return f"<div class='comment'>{content}</div>"
    
    def add_comment_safe(self, content: str) -> str:
        """安全版本：转义 HTML 特殊字符"""
        # escape 会把 <script> 变成 &lt;script&gt;
        safe_content = escape(content)
        self.comments.append(safe_content)
        return f"<div class='comment'>{safe_content}</div>"

# 使用示例
if __name__ == "__main__":
    system = CommentSystem()
    
    # 攻击者提交的恶意内容
    malicious_input = "<script>alert('偷走你的cookie!')</script>"
    
    # 不安全版本 - 恶意脚本会直接执行
    unsafe_output = system.add_comment_unsafe(malicious_input)
    print("不安全输出:")
    print(unsafe_output)
    
    # 安全版本 - 脚本被转义，不会执行
    safe_output = system.add_comment_safe(malicious_input)
    print("\n安全输出:")
    print(safe_output)

# 预期输出：
# 不安全输出:
# <div class='comment'><script>alert('偷走你的cookie!')</script></div>
# 
# 安全输出:
# <div class='comment'>&lt;script&gt;alert('偷走你的cookie!')&lt;/script&gt;</div>
```

## 示例 3：完整的 Flask 认证系统

```python
# 一个简单的 Flask 认证系统（使用 SQLite）
from flask import Flask, request, jsonify, session
from functools import wraps
import hashlib
import secrets
import sqlite3
import os

app = Flask(__name__)
# 生产环境必须使用随机密钥！
app.secret_key = os.environ.get('SECRET_KEY', 'dev-secret-key')

def get_db():
    """获取数据库连接"""
    conn = sqlite3.connect('users.db')
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    """初始化数据库表"""
    conn = get_db()
    conn.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            salt TEXT NOT NULL,
            password_hash TEXT NOT NULL
        )
    ''')
    conn.commit()
    conn.close()

def hash_password(password: str) -> tuple:
    """生成密码哈希和盐值"""
    salt = secrets.token_hex(16)
    hash_value = hashlib.pbkdf2_hmac(
        'sha256', password.encode(), salt.encode(), 100000
    ).hex()
    return salt, hash_value

def verify_password(password: str, salt: str, expected_hash: str) -> bool:
    """验证密码"""
    hash_value = hashlib.pbkdf2_hmac(
        'sha256', password.encode(), salt.encode(), 100000
    ).hex()
    return hash_value == expected_hash

def login_required(f):
    """登录验证装饰器"""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'user_id' not in session:
            return jsonify({"error": "请先登录"}), 401
        return f(*args, **kwargs)
    return decorated_function

@app.route('/register', methods=['POST'])
def register():
    """用户注册"""
    data = request.get_json()
    username = data.get('username')
    password = data.get('password')
    
    if not username or not password:
        return jsonify({"error": "用户名和密码不能为空"}), 400
    
    salt, hash_val = hash_password(password)
    
    conn = get_db()
    try:
        conn.execute(
            'INSERT INTO users (username, salt, password_hash) VALUES (?, ?, ?)',
            (username, salt, hash_val)
        )
        conn.commit()
        return jsonify({"message": "注册成功"}), 201
    except sqlite3.IntegrityError:
        return jsonify({"error": "用户名已存在"}), 400
    finally:
        conn.close()

@app.route('/login', methods=['POST'])
def login():
    """用户登录"""
    data = request.get_json()
    username = data.get('username')
    password = data.get('password')
    
    conn = get_db()
    user = conn.execute(
        'SELECT * FROM users WHERE username = ?', (username,)
    ).fetchone()
    conn.close()
    
    if user and verify_password(password, user['salt'], user['password_hash']):
        session['user_id'] = user['id']
        session['username'] = user['username']
        return jsonify({"message": "登录成功"})
    
    return jsonify({"error": "用户名或密码错误"}), 401

@app.route('/profile')
@login_required
def profile():
    """受保护的用户信息接口"""
    return jsonify({
        "username": session['username'],
        "message": "这是只有登录用户才能看到的信息"
    })

@app.route('/logout', methods=['POST'])
def logout():
    """退出登录"""
    session.clear()
    return jsonify({"message": "已退出登录"})

if __name__ == "__main__":
    init_db()
    # 测试代码
    with app.test_client() as client:
        # 注册用户
        response = client.post('/register', json={
            "username": "alice",
            "password": "secure_pass_123"
        })
        print(f"注册: {response.get_json()}")
        
        # 登录
        response = client.post('/login', json={
            "username": "alice",
            "password": "secure_pass_123"
        })
        print(f"登录: {response.get_json()}")
        
        # 访问受保护接口
        response = client.get('/profile')
        print(f"个人信息: {response.get_json()}")
        
        # 退出登录
        response = client.post('/logout')
        print(f"退出: {response.get_json()}")
        
        # 退出后访问受保护接口
        response = client.get('/profile')
        print(f"退出后访问: {response.get_json()}")

# 预期输出：
# 注册: {'message': '注册成功'}
# 登录: {'message': '登录成功'}
# 个人信息: {'username': 'alice', 'message': '这是只有登录用户才能看到的信息'}
# 退出: {'message': '已退出登录'}
# 退出后访问: {'error': '请先登录'}
```

## 常用函数/调用速查表

| 函数 | 签名 | 用途 | 最小示例 |
|------|------|------|---------|
| `escape` | `escape(text: str) -> str` | 转义 HTML 特殊字符 | `escape("<script>")` → `"&lt;script&gt;"` |
| `hashlib.pbkdf2_hmac` | `pbkdf2_hmac(name, password, salt, iterations)` | 安全的密码哈希 | `pbkdf2_hmac('sha256', b'pass', b'salt', 100000)` |
| `secrets.token_hex` | `token_hex(nbytes: int) -> str` | 生成随机十六进制字符串 | `token_hex(16)` → `"3f9a2b..."` |
| `secrets.compare_digest` | `compare_digest(a, b) -> bool` | 常数时间比较（防时序攻击） | `compare_digest(hash1, hash2)` |
| `session.clear` | `clear() -> None` | 清除所有会话数据 | `session.clear()` |
| `@wraps` | `wraps(func)` | 保留函数元信息 | `@wraps(f)` 在装饰器中使用 |

## 5个新手易错点

### 易错点1：密码明文存储

❌ **错误写法**
```python
# 危险！密码明文存储在数据库中
import sqlite3

# 创建数据库
conn = sqlite3.connect('bad_users.db')
conn.execute('''
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY,
        username TEXT,
        password TEXT  # 直接存明文密码！
    )
''')

# 注册用户（明文存储密码）
def register_user(username, password):
    conn.execute(
        'INSERT INTO users (username, password) VALUES (?, ?)',
        (username, password)  # 密码直接存进去
    )
    conn.commit()
    print(f"用户 {username} 注册成功，密码: {password}")

# 测试
register_user("bob", "my_secret_123")
print("数据库内容:")
for row in conn.execute('SELECT * FROM users'):
    print(f"  用户名: {row[1]}, 密码: {row[2]}")  # 密码直接可见！

# 运行结果：密码明文显示，数据库泄露=所有账号泄露
```

✅ **正确写法**
```python
# 安全！使用哈希存储密码
import hashlib
import secrets
import sqlite3

conn = sqlite3.connect('good_users.db')
conn.execute('''
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY,
        username TEXT,
        salt TEXT,
        password_hash TEXT
    )
''')

def hash_password(password):
    """生成盐值和哈希"""
    salt = secrets.token_hex(16)
    hash_val = hashlib.pbkdf2_hmac(
        'sha256', password.encode(), salt.encode(), 100000
    ).hex()
    return salt, hash_val

def register_user(username, password):
    salt, hash_val = hash_password(password)
    conn.execute(
        'INSERT INTO users (username, salt, password_hash) VALUES (?, ?, ?)',
        (username, salt, hash_val)
    )
    conn.commit()
    print(f"用户 {username} 注册成功")

# 测试
register_user("bob", "my_secret_123")
print("数据库内容:")
for row in conn.execute('SELECT * FROM users'):
    print(f"  用户名: {row[1]}, 盐值: {row[2][:8]}..., 哈希: {row[3][:8]}...")
    # 密码不可见，只有哈希值！

# 运行结果：密码被哈希，即使数据库泄露也无法获取原始密码
```

**为什么错**：明文密码一旦数据库泄露，所有用户账号立即暴露。哈希是单向的，即使拿到哈希值也无法反推原始密码。

### 易错点2：直接拼接用户输入到 SQL 查询

❌ **错误写法**
```python
# 危险！SQL 注入漏洞
import sqlite3

conn = sqlite3.connect('inject_demo.db')
conn.execute('CREATE TABLE IF NOT EXISTS products (id INTEGER, name TEXT)')
conn.execute("INSERT INTO products VALUES (1, '苹果')")
conn.execute("INSERT INTO products VALUES (2, '香蕉')")
conn.commit()

# 不安全的查询 - 直接拼接用户输入
def search_products(user_input):
    query = f"SELECT * FROM products WHERE name = '{user_input}'"
    print(f"执行的SQL: {query}")
    result = conn.execute(query).fetchall()
    return result

# 攻击者输入恶意内容
malicious_input = "' OR '1'='1"
results = search_products(malicious_input)
print(f"查询结果: {results}")
# 攻击者绕过了条件，返回所有商品！

# 运行结果：攻击者通过注入 OR '1'='1' 获取了所有数据
```

✅ **正确写法**
```python
# 安全！使用参数化查询
import sqlite3

conn = sqlite3.connect('safe_demo.db')
conn.execute('CREATE TABLE IF NOT EXISTS products (id INTEGER, name TEXT)')
conn.execute("INSERT INTO products VALUES (1, '苹果')")
conn.execute("INSERT INTO products VALUES (2, '香蕉')")
conn.commit()

# 安全的查询 - 使用参数占位符
def search_products(user_input):
    query = "SELECT * FROM products WHERE name = ?"
    print(f"执行的SQL: {query} (参数: {user_input})")
    # 参数会被自动转义，不会执行恶意SQL
    result = conn.execute(query, (user_input,)).fetchall()
    return result

# 即使输入恶意内容，也只是当作普通字符串
malicious_input = "' OR '1'='1"
results = search_products(malicious_input)
print(f"查询结果: {results}")  # 返回空列表，因为找不到匹配的商品

# 正常查询
normal_results = search_products("苹果")
print(f"正常查询结果: {normal_results}")

# 运行结果：恶意输入被当作普通字符串处理，无法注入SQL
```

**为什么错**：直接拼接用户输入到 SQL 中，攻击者可以注入任意 SQL 代码。参数化查询让数据库把输入当作数据而非代码。

### 易错点3：忘记转义用户输入（XSS）

❌ **错误写法**
```python
# 危险！XSS 漏洞
from flask import Flask, request

app = Flask(__name__)

@app.route('/comment', methods=['POST'])
def add_comment():
    # 直接使用用户输入，没有转义
    user_input = request.form['content']
    # 直接把用户输入拼进 HTML
    html = f"<div class='comment'>{user_input}</div>"
    return html

# 攻击者提交: <script>alert('攻击!')</script>
# 浏览器会执行这个脚本，可以窃取cookie、篡改页面等
# 运行结果：恶意脚本在用户浏览器中执行
```

✅ **正确写法**
```python
# 安全！转义用户输入
from flask import Flask, request
from html import escape

app = Flask(__name__)

@app.route('/comment', methods=['POST'])
def add_comment():
    # 获取用户输入
    user_input = request.form['content']
    # 转义 HTML 特殊字符
    safe_input = escape(user_input)
    # 使用转义后的内容
    html = f"<div class='comment'>{safe_input}</div>"
    return html

# 攻击者提交: <script>alert('攻击!')</script>
# 转义后变成: &lt;script&gt;alert('攻击!')&lt;/script&gt;
# 浏览器显示为文本，不会执行脚本
# 运行结果：恶意脚本被当作普通文本显示，不会执行
```

**为什么错**：不转义用户输入，攻击者可以注入任意 HTML/JavaScript 代码。`escape()` 函数把 `<`、`>` 等特殊字符转义，让浏览器把它们当作文本而非代码。

### 易错点4：使用不安全的随机数生成器

❌ **错误写法**
```python
# 危险！使用不安全的随机数
import random

def generate_session_token():
    """生成会话令牌"""
    # random 模块不是密码学安全的！
    # 攻击者可能预测生成的令牌
    return str(random.randint(100000, 999999))

# 测试
tokens = []
for _ in range(5):
    token = generate_session_token()
    tokens.append(token)
    print(f"生成的令牌: {token}")

# 使用 random 模块，攻击者可能通过分析预测下一个令牌
# 运行结果：令牌可预测，攻击者可能伪造会话
```

✅ **正确写法**
```python
# 安全！使用密码学安全的随机数
import secrets

def generate_session_token():
    """生成会话令牌"""
    # secrets 模块是密码学安全的
    # 生成的令牌不可预测
    return secrets.token_hex(16)

# 测试
tokens = []
for _ in range(5):
    token = generate_session_token()
    tokens.append(token)
    print(f"生成的令牌: {token[:16]}...")

# 使用 secrets 模块，每个令牌都是随机且不可预测的
# 运行结果：令牌完全随机，攻击者无法预测
```

**为什么错**：`random` 模块是伪随机数生成器，用于模拟和游戏，不适合安全场景。`secrets` 模块使用操作系统提供的加密安全随机源。

### 易错点5：在生产环境使用调试模式

❌ **错误写法**
```python
# 危险！生产环境开启调试模式
from flask import Flask
```

✅ **正确写法**
```python
# 安全！生产环境关闭调试模式
from flask import Flask
import os

app = Flask(__name__)

@app.route('/')
def index():
    return "Hello World"

if __name__ == "__main__":
    # 通过环境变量控制调试模式
    debug_mode = os.environ.get('FLASK_DEBUG', 'False').lower() == 'true'
    
    if debug_mode:
        # 开发环境：开启调试
        app.run(debug=True, host='127.0.0.1', port=5000)
    else:
        # 生产环境：关闭调试，只监听本地
        app.run(debug=False, host='0.0.0.0', port=5000)
    
    # 运行结果：生产环境不暴露错误信息，更安全
```

**为什么错**：调试模式会显示详细的错误堆栈、交互式调试器，攻击者可以利用这些信息发现系统漏洞。生产环境必须关闭调试模式。

```python
import hashlib
import secrets

class User:
    def __init__(self, username):
        self.username = username
        self.salt = None
        self.password_hash = None
    
    def set_password(self, password):
        # 生成随机盐值
        self.salt = secrets.token_hex(16)
        # 计算密码哈希
        self.password_hash = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode(),
            self.salt.encode(),
            100000
        ).hex()
    
    def check_password(self, password):
        # 如果没有设置密码，返回 False
        if not self.salt or not self.password_hash:
            return False
        # 重新计算哈希并比较
        hash_value = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode(),
            self.salt.encode(),
            100000
        ).hex()
        return hash_value == self.password_hash
```

```python
import hashlib
import secrets

class UserManager:
    def __init__(self):
        # 使用字典存储用户: {username: {'salt': ..., 'hash': ...}}
        self.users = {}
    
    def register(self, username, password):
        # 检查用户名是否已存在
        if username in self.users:
            return "用户名已存在"
        
        # 生成盐值和哈希
        salt = secrets.token_hex(16)
        hash_value = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode(),
            salt.encode(),
            100000
        ).hex()
        
        # 存储用户信息
        self.users[username] = {
            'salt': salt,
            'hash': hash_value
        }
        return "注册成功"
    
    def login(self, username, password):
        # 检查用户是否存在
        if username not in self.users:
            return "用户不存在"
        
        # 获取用户信息
        user = self.users[username]
        
        # 验证密码
        hash_value = hashlib.pbkdf2_hmac(
            'sha256',
            password.encode(),
            user['salt'].encode(),
            100000
        ).hex()
        
        if hash_value == user['hash']:
            return "登录成功"
        else:
            return "用户名或密码错误"

print(manager.register("bob", "secure456"))

print(manager.login("charlie", "password123"))
```
